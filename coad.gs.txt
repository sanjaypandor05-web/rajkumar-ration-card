// ---------- SPREADSHEET ----------

const SPREADSHEET_ID = "YOUR_SPREADSHEET_ID";

const ss = SpreadsheetApp.openById(SPREADSHEET_ID);

// ---------- SHEETS ----------

const APPLICATION_SHEET = "Applications";

// ---------- DRIVE FOLDER ----------

const DOCUMENT_FOLDER_ID = "YOUR_DRIVE_FOLDER_ID";

const folder = DriveApp.getFolderById(DOCUMENT_FOLDER_ID);

// ==========================================
// GENERATE APPLICATION ID
// ==========================================

function generateApplicationId(){

  const sheet = ss.getSheetByName(APPLICATION_SHEET);

  const lastRow = sheet.getLastRow();

  if(lastRow <= 1){

    return "RC202600001";

  }

  const lastId = sheet.getRange(lastRow,1).getValue();

  const number = Number(lastId.replace("RC",""));

  const next = number + 1;

  return "RC" + next.toString().padStart(9,"0");

}

// ==========================================
// POST REQUEST
// ==========================================

function doPost(e){

  try{

    const data = JSON.parse(e.postData.contents);

    const applicationId = generateApplicationId();

    // આગળ Part 2 માં આવશે

    return ContentService
      .createTextOutput(JSON.stringify({
        success:true,
        applicationId:applicationId
      }))
      .setMimeType(ContentService.MimeType.JSON);

  }catch(err){

    return ContentService
      .createTextOutput(JSON.stringify({
        success:false,
        message:err.toString()
      }))
      .setMimeType(ContentService.MimeType.JSON);

  }

}

// ---------- BASE64 TO DRIVE ----------

function saveBase64File(base64Data, fileName) {

  if (!base64Data) return "";

  const matches = base64Data.match(/^data:(.+);base64,(.+)$/);

  if (!matches) return "";

  const mimeType = matches[1];
  const bytes = Utilities.base64Decode(matches[2]);

  const blob = Utilities.newBlob(bytes, mimeType, fileName);

  const file = folder.createFile(blob);

  return file.getUrl();

}

// ==========================================
// SAVE APPLICATION
// ==========================================

function saveApplication(data, applicationId){

  const sheet = ss.getSheetByName(APPLICATION_SHEET);

  const aadharPdfUrl = saveBase64File(
    data.aadharPdf,
    applicationId + "_Aadhaar.pdf"
  );

  const rationPdfUrl = saveBase64File(
    data.rationPdf,
    applicationId + "_RationCard.pdf"
  );

  const otherPdfUrl = saveBase64File(
    data.otherPdf,
    applicationId + "_Other.pdf"
  );

  const paymentUrl = saveBase64File(
    data.paymentScreenshot,
    applicationId + "_Payment.jpg"
  );

  sheet.appendRow([

    applicationId,

    new Date(),

    data.service,

    data.applicantName,

    data.relation,

    data.relationName,

    data.mobile,

    data.aadhar,

    data.ration,

    data.village,

    data.taluka,

    data.district,

    data.details,

    aadharPdfUrl,

    rationPdfUrl,

    otherPdfUrl,

    paymentUrl,

    "Pending"

  ]);

}
